cmake_minimum_required(VERSION 3.20)
project(jit_matvec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ONEDNN_SOURCE "" CACHE PATH "Pinned oneDNN v3.9.1 source (Xbyak assembler only)")
set(XBYAK "${ONEDNN_SOURCE}/third_party/xbyak_aarch64")
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  message(FATAL_ERROR "This experiment currently requires Apple Silicon")
endif()
add_library(assembler STATIC ${XBYAK}/src/xbyak_aarch64_impl.cpp ${XBYAK}/src/util_impl.cpp)
target_include_directories(assembler PUBLIC ${XBYAK} PRIVATE ${XBYAK}/xbyak_aarch64)
add_executable(jit_matvec main.cpp baselines.cpp)
target_compile_options(jit_matvec PRIVATE -O3 -Wall -Wextra -mcpu=apple-m4)
target_link_libraries(jit_matvec PRIVATE assembler "-framework Accelerate")
