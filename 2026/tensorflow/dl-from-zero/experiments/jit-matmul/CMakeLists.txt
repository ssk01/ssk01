cmake_minimum_required(VERSION 3.20)
project(libxsmm_matmul_case LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(LIBXSMM_ROOT "" CACHE PATH "Built LIBXSMM source tree")
add_executable(jit_matmul main.cpp)
target_include_directories(jit_matmul PRIVATE ${LIBXSMM_ROOT}/include)
target_compile_definitions(jit_matmul PRIVATE ACCELERATE_NEW_LAPACK)
target_compile_options(jit_matmul PRIVATE -O3 -Wall -Wextra)
target_link_libraries(jit_matmul PRIVATE ${LIBXSMM_ROOT}/lib/libxsmm.a "-framework Accelerate")
