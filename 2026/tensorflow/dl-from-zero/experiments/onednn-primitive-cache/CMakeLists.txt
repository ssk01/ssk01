cmake_minimum_required(VERSION 3.16)
project(primitive_cache_case LANGUAGES CXX)
find_package(dnnl CONFIG REQUIRED)
add_executable(cache_case main.cpp)
target_compile_features(cache_case PRIVATE cxx_std_17)
target_compile_options(cache_case PRIVATE -Wall -Wextra)
target_link_libraries(cache_case PRIVATE DNNL::dnnl)
add_executable(arm_jit_probe arm_jit_probe.cpp)
target_compile_features(arm_jit_probe PRIVATE cxx_std_17)
target_compile_options(arm_jit_probe PRIVATE -Wall -Wextra)
target_link_libraries(arm_jit_probe PRIVATE DNNL::dnnl)
