cmake_minimum_required(VERSION 3.20)
project(primitive_cache LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(LIBXSMM_ROOT "" CACHE PATH "Dereg-enabled LIBXSMM source tree; empty disables JIT")
add_executable(primitive_demo demo.cpp cpu_only.cpp)
target_compile_definitions(primitive_demo PRIVATE ACCELERATE_NEW_LAPACK)
target_compile_options(primitive_demo PRIVATE -O3)
target_link_libraries(primitive_demo PRIVATE "-framework Accelerate")
if(LIBXSMM_ROOT)
 target_compile_definitions(primitive_demo PRIVATE LF_LIBXSMM)
 target_include_directories(primitive_demo PRIVATE ${LIBXSMM_ROOT}/include)
 target_link_libraries(primitive_demo PRIVATE ${LIBXSMM_ROOT}/lib/libxsmm.a)
endif()
enable_testing()
add_test(NAME correctness COMMAND primitive_demo test)
set_tests_properties(correctness PROPERTIES ENVIRONMENT "VECLIB_MAXIMUM_THREADS=1")
